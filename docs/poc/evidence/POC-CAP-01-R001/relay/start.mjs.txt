import fs from 'node:fs';
import { monitorEventLoopDelay } from 'node:perf_hooks';
import { createApp } from '../../dist/server/app.js';
const cfg=JSON.parse(fs.readFileSync('.scratch/cap01/runtime-private.json','utf8'));
const db=new URL(cfg.databaseUrl); db.hostname='host.docker.internal';
const app=await createApp({https:{cert:fs.readFileSync('.scratch/t01/tls/localhost-cert.pem'),key:fs.readFileSync('.scratch/t01/tls/localhost-key.pem')},databaseUrl:db.href,hostControlToken:'',meetingRoomPassword:''});
const loop=monitorEventLoopDelay({resolution:20}); loop.enable();
await app.listen({host:'0.0.0.0',port:4173});
console.log(JSON.stringify({kind:'ready',node:process.version,arch:process.arch,persistence:'postgresql'}));
let cpu=process.cpuUsage(), last=performance.now();
const timer=setInterval(()=>{const now=performance.now(),next=process.cpuUsage();console.log(JSON.stringify({kind:'sample',at:Date.now(),elapsedMs:now-last,cpuPercent:100*(next.user+next.system-cpu.user-cpu.system)/1000/(now-last),...process.memoryUsage(),eventLoopP95Ms:loop.percentile(95)/1e6,eventLoopMaxMs:loop.max/1e6}));cpu=next;last=now;loop.reset();},1000);
for(const signal of ['SIGTERM','SIGINT'])process.once(signal,async()=>{clearInterval(timer);loop.disable();await app.close();process.exit(0);});
