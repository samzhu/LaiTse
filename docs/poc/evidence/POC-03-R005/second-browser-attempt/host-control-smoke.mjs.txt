function redact(text) {
  let value = String(text ?? "");

}

function safeMessage(payload, direction) {
  let message;
  try { message = JSON.parse(payload); } catch { return { direction, type: "unparsed-control" }; }
  const record = { direction, type: typeof message?.type === "string" ? message.type : "unknown" };
  for (const key of ["roomId", "currentRoomId", "action", "code", "reason"]) {
    if (typeof message?.[key] === "string") record[key] = message[key];
  }
  if (message.type === "room.snapshot") {
    record.roomId = message.room?.id ?? "";
    record.isLocked = Boolean(message.room?.isLocked);
    record.hostControlsEnabled = Boolean(message.room?.hostControlsEnabled);
    record.publishAllowed = Boolean(message.room?.publishAllowed);
    record.memberCount = Array.isArray(message.members) ? message.members.length : 0;
  }
  if (message.type === "space.snapshot") {
    record.currentRoomId = message.currentRoomId;
    record.memberCount = message.rooms?.find((room) => room.id === message.currentRoomId)?.players?.length ?? 0;
  }
  if (message.type === "host.publishPermission") {
    record.enabled = Boolean(message.enabled);
    record.allowed = Boolean(message.allowed);
  }
  if (message.type === "host.actionResult") {
    for (const key of ["isLocked", "allowed", "membersRetained", "grantsCleared"]) {
      if (typeof message[key] === "boolean") record[key] = message[key];
    }
  }
  if (message.type === "host.error" || message.type === "protocol.error") record.code = message.code ?? "unknown";
  return record;
}

function attachPage(page, name) {
  const client = { name, controls: [], packets: { sent: { audio: 0, screen: 0 }, received: { audio: 0, screen: 0 } }, errors: [], latestSpaceSnapshot: null };
  page.on("pageerror", (error) => {
    const item = redact(error.stack ?? error.message);
    client.errors.push(item);
    report.errors.pageErrors.push({ client: name, message: item });
  });
  page.on("websocket", (socket) => {
    socket.on("framesent", (frame) => recordFrame(client, frame, "sent"));
    socket.on("framereceived", (frame) => recordFrame(client, frame, "received"));
    socket.on("socketerror", (error) => report.errors.socketErrors.push({ client: name, message: redact(error) }));
  });
  clients[name] = { page, client };
  return client;
}

function recordFrame(client, frame, direction) {
  if (typeof frame.payload !== "string") {
    const bytes = Buffer.isBuffer(frame.payload) ? frame.payload : Buffer.from(frame.payload);
    const magic = bytes.toString("ascii", 0, 4);
    if (magic === "LTA1") client.packets[direction].audio += 1;
    if (magic === "LTS1") client.packets[direction].screen += 1;
    return;
  }
  const record = safeMessage(frame.payload, direction);
  if (record.type === "space.snapshot") {
    try { client.latestSpaceSnapshot = JSON.parse(frame.payload); } catch { client.latestSpaceSnapshot = null; }
  }
  if (client.controls.length < 500) client.controls.push(record);
}

async function installSyntheticSources(page, frequency = 440) {
  await page.addInitScript(({ frequency }) => {
    const state = { frequency, context: null, tracks: [] };
    Object.defineProperty(window, "__t09Synthetic", { value: state });
    const devices = navigator.mediaDevices;
    Object.defineProperty(devices, "getUserMedia", {
      configurable: true,
      value: async () => {
        const context = new AudioContext({ sampleRate: 48_000 });
        await context.resume();
        const destination = context.createMediaStreamDestination();
        const oscillator = context.createOscillator();
        const gain = context.createGain();
        oscillator.frequency.value = state.frequency;
        gain.gain.value = 0.22;
        oscillator.connect(gain).connect(destination);
        oscillator.start();
        state.context = context;
        state.tracks.push(...destination.stream.getTracks());
        return destination.stream;
      },
    });
    Object.defineProperty(devices, "getDisplayMedia", {
      configurable: true,
      value: async () => {
        const canvas = document.createElement("canvas");
        canvas.width = 320;
        canvas.height = 180;
        const context = canvas.getContext("2d");
        if (!context) throw new Error("Synthetic screen source could not create a canvas context.");
        let running = true;
        const draw = () => {
          if (!running) return;
          context.fillStyle = "#265c8c";
          context.fillRect(0, 0, canvas.width, canvas.height);
          context.fillStyle = "#f4d35e";
          context.fillRect(12, 12, 48, 14);
          requestAnimationFrame(draw);
        };
        draw();
        const stream = canvas.captureStream(10);
        const track = stream.getVideoTracks()[0];
        if (!track) throw new Error("Synthetic screen source did not return a video track.");
        state.tracks.push(track);
        const stop = track.stop.bind(track);
        track.stop = () => { running = false; stop(); };
        return stream;
      },
    });
  }, { frequency });
}

}

async function waitFor(predicate, description, timeoutMs = 8_000) {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    const value = predicate();
    if (value) return value;
    await new Promise((resolveWait) => setTimeout(resolveWait, 25));
  }
  throw new Error(`Timed out waiting for ${description}.`);
}

function controlCount(client, type, direction = "received") {
  return client.controls.filter((entry) => entry.type === type && entry.direction === direction).length;
}

async function waitForControl(client, type, previous = 0, timeoutMs = 8_000) {
  return waitFor(() => client.controls.filter((entry) => entry.type === type && entry.direction === "received")[previous] ?? null, `${client.name} ${type}`, timeoutMs);
}

function probeJson(socket, predicate, timeoutMs = 5_000) {
  return new Promise((resolve, reject) => {
    const timeout = setTimeout(() => { socket.off("message", onMessage); reject(new Error("WSS probe response timed out.")); }, timeoutMs);
    const onMessage = (raw, isBinary) => {
      if (isBinary) return;
      const message = JSON.parse(raw.toString());
      if (!predicate(message)) return;
      clearTimeout(timeout);
      socket.off("message", onMessage);
      resolve(message);
    };
    socket.on("message", onMessage);
  });
}

async function runUnauthorizedProbe() {
  const socket = new WebSocket(`${baseWSURL()}/ws`, { origin: target.origin, ca: caPem });
  probeSockets.add(socket);
  await new Promise((resolve, reject) => {
    socket.once("open", resolve);
    socket.once("error", reject);
  });
  const joined = probeJson(socket, (message) => message.type === "room.snapshot");
  socket.send(JSON.stringify({ type: "join", roomId: "studio", displayName: "Protocol probe" }));
  await joined;

  const attempts = [
    { type: "host.grantPublisher", participantId: "forged-member", role: "host" },
    { type: "host.revokePublisher", participantId: "forged-member", role: "host" },
    { type: "host.removeParticipant", participantId: "forged-member", role: "host" },
    { type: "host.setMeetingRoomPassword", password: "ignored-probe-value", role: "host" },
    { type: "host.endSession", role: "host" },
  ];
  const outcomes = [];
  for (const attempt of attempts) {
    const response = probeJson(socket, (message) => message.type === "host.error");
    socket.send(JSON.stringify(attempt));
    const received = await response;
    assert.equal(received.code, "host_unauthorized");
    outcomes.push({ operation: attempt.type, code: received.code });
  }

  const audioDenied = probeJson(socket, (message) => message.type === "audio.error");
  socket.send(JSON.stringify({ type: "audio.start", codec: "opus", sampleRate: 48_000, channels: 1, bitrate: 32_000, frameDurationUs: 20_000 }));
  assert.equal((await audioDenied).code, "publish_not_authorized");
  const screenDenied = probeJson(socket, (message) => message.type === "screen.error");
  socket.send(JSON.stringify({ type: "screen.start", codec: "vp8", width: 320, height: 180, frameRate: 15, bitrate: 800_000 }));
  assert.equal((await screenDenied).code, "publish_not_authorized");

  const binaryDenied = probeJson(socket, (message) => message.type === "host.publishDenied");
  const frame = Buffer.from([0x4c, 0x54, 0x41, 0x31, 0, 0, 0, 1, 0, 0, 0, 0]);
  socket.send(frame);
  await binaryDenied;
  const screenBinaryDenied = probeJson(socket, (message) => message.type === "host.publishDenied");
  socket.send(Buffer.from([0x4c, 0x54, 0x53, 0x31, 0, 0, 0, 1, 0, 0, 0, 0]));
  await screenBinaryDenied;
  const diagnostics = await stageProbe();
  assert.equal(diagnostics.isLocked, false, "unauthorized lock attempts must not change the meeting-room gate");
  socket.close();
  probeSockets.delete(socket);
  report.maliciousOperations = outcomes;
  return { deniedAudioStart: true, deniedScreenStart: true, deniedAudioBinary: true, deniedScreenBinary: true };
}

async function waitForMedia(page, client) {
  await page.waitForFunction(() => {
    const audio = document.querySelector('[data-testid="audio-panel"]');
    const screen = document.querySelector('[data-testid="screen-panel"]');
    return Number(audio?.getAttribute("data-rendered-samples")) > 0 && Number(screen?.getAttribute("data-rendered-frames")) > 2;
  }, undefined, { timeout: 10_000 });
  await waitFor(() => client.packets.sent.audio > 0 && client.packets.sent.screen > 0, `${client.name} sends synthetic audio and screen`);
  return await page.evaluate(() => ({
    audioRenderedSamples: Number(document.querySelector('[data-testid="audio-panel"]')?.getAttribute("data-rendered-samples")),
    screenRenderedFrames: Number(document.querySelector('[data-testid="screen-panel"]')?.getAttribute("data-rendered-frames")),
  }));
}

function writeReport() {
  report.finishedAt = new Date().toISOString();
  report.status = failure ? "failed" : "passed";
  report.errors.failure = failure ? redact(failure.stack ?? failure) : null;
  for (const [name, item] of Object.entries(clients)) {
    report.clients[name] = {
      controls: item.client.controls,
      packets: item.client.packets,
      pageErrors: item.client.errors,
      latestSpace: item.client.latestSpaceSnapshot ? {
        currentRoomId: item.client.latestSpaceSnapshot.currentRoomId,
        memberCount: item.client.latestSpaceSnapshot.rooms?.find((room) => room.id === item.client.latestSpaceSnapshot.currentRoomId)?.players?.length ?? 0,
      } : null,
    };
  }
  const serialized = JSON.stringify(report);
  for (const secret of [hostToken, meetingPassword, wrongPassword]) {
    if (serialized.includes(secret)) throw new Error("T09 report unexpectedly contains a credential value.");
  }
  writeFileSync(join(outputDirectory, "report.json"), `${JSON.stringify(report, null, 2)}\n`);
  writeFileSync(join(outputDirectory, "summary.txt"), [
    `Status: ${report.status}`,
    `Started: ${report.startedAt}`,
    `Finished: ${report.finishedAt}`,
    `Clients: ${Object.keys(clients).join(", ")}`,
    `Checks: ${report.checks.map((item) => item.name).join(", ")}`,
    `Screenshots: ${report.screenshots.join(", ") || "none"}`,
    "Synthetic desktop evidence only; the report does not store host or room passwords.",
  ].join("\n") + "\n");
}
